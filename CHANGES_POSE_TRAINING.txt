// so class 10+ never collides with an earlier class's color.
function hueToHex(h, s=70, l=50){
    s/=100; l/=100;
    const k = n => (n + h/30) % 12;
    const a = s * Math.min(l, 1-l);
    const f = n => l - a * Math.max(-1, Math.min(k(n)-3, Math.min(9-k(n),1)));
    const toHex = x => Math.round(255*x).toString(16).padStart(2,'0');
    return `#${toHex(f(0))}${toHex(f(8))}${toHex(f(4))}`;
}
function colorForId(id){

--------------------------------------------------------------------------
CHANGE 1.4 — Pose features were position-dependent, not shape-dependent
--------------------------------------------------------------------------
PROBLEM: keypointsToFeatures() normalized each keypoint by dividing by
frame width/height only. That means the model partially learned WHERE in
the frame you stood, not the SHAPE of the pose. As more classes were
added, this caused confusion between classes captured at slightly
different distances/positions from the camera.

FIX: center coordinates on the hip midpoint (keypoints 11,12) and scale
by torso length (shoulder midpoint to hip midpoint distance, keypoints
5,6 vs 11,12). This makes features translation/scale invariant.

IMPORTANT: this changes the feature format. Any previously trained model
must be retrained after this change — old dataset samples are still
valid (they store raw MoveNet keypoints... actually they store the
already-computed feature vector, see note below), but a model trained
with the OLD format will not work with NEW-format live inference, and
vice versa. See CHANGE 1.4-NOTE below.

LOCATION: keypointsToFeatures() function (~line 804-833).

// ──────────────────────────────────────────────────────────────
// MOVENET FEATURE EXTRACTION
// Flattens 17 keypoints × (x, y) → Float32Array of length 34.
// Coordinates are centered on the hip midpoint and scaled by torso
// length (shoulder-to-hip distance) so the features describe pose
// *shape* rather than where the person stands in the frame. Without
// this, classes captured at slightly different distances/positions
// overlap in feature space and become harder to tell apart as more
// classes are added.
// ──────────────────────────────────────────────────────────────
function keypointsToFeatures(keypoints, imgW, imgH){
    const arr = new Float32Array(NUM_KEYPOINTS * 2);

    const lHip = keypoints[11], rHip = keypoints[12];
    const lSh  = keypoints[5],  rSh  = keypoints[6];

    const cx = (lHip && rHip) ? (lHip.x + rHip.x) / 2 : imgW / 2;
    const cy = (lHip && rHip) ? (lHip.y + rHip.y) / 2 : imgH / 2;
    const shx = (lSh && rSh) ? (lSh.x + rSh.x) / 2 : cx;
    const shy = (lSh && rSh) ? (lSh.y + rSh.y) / 2 : cy - imgH * 0.25;

    let scale = Math.hypot(shx - cx, shy - cy);
    if (!scale || scale < 1e-3) scale = Math.max(imgW, imgH) * 0.25;

    for(let i = 0; i < NUM_KEYPOINTS; i++){
        const kp = keypoints[i];
        arr[i*2]   = kp ? (kp.x - cx) / scale : 0;
        arr[i*2+1] = kp ? (kp.y - cy) / scale : 0;
    }
    return arr;
}

CHANGE 1.4-NOTE: keypointsToFeatures() is called once at CAPTURE time
(in captureOneSample) and the resulting feature vector is what gets
stored in `dataset`. It is also called at PREDICT time (in
_startPoseInferLoop). Both call sites automatically pick up whichever
version of this function is loaded on the page, so as long as you
retrain AFTER this edit (not before), train and predict stay consistent
with each other. You do not need to touch captureOneSample() or
_startPoseInferLoop() — they just call keypointsToFeatures() and don't
care about its internals.

        SERIAL IMAGE_END\n
    or on failure: "SERIAL ERR <message>\n"
  - "Live" view is faked by polling CAPTURE on a timer (we use 1000ms).

// ──────────────────────────────────────────────────────────────
// ESP32-S3 CAMERA (view-only, over WebSerial)
// Firmware has no continuous video stream — it replies to a
// "CAPTURE\n" command with one base64 JPEG snapshot framed as:
//   SERIAL IMAGE_BEGIN <len> <b64len>\n
//   SERIAL IMAGE_CHUNK <b64 chars>\n   (repeated)
//   SERIAL IMAGE_END\n
// so "live" view is simulated by polling CAPTURE on a timer.
// Frames are shown for visual confirmation only — they are NOT fed
// into MoveNet, since the 96×96 grayscale sensor is far too small
// for reliable body-keypoint detection. Training samples still come
// from the regular Webcam button.
// ──────────────────────────────────────────────────────────────
async function openEsp32CamCapture(classIdx){
    if(!('serial' in navigator)){alert('ESP32-S3 USB camera requires Chrome / Edge v89+ on desktop.');return;}
    if(camOpenIdx !== null) stopCamCapture();

    try{
        if(!esp32Port){
            esp32Port = await navigator.serial.requestPort();
            await esp32Port.open({baudRate:115200});
            esp32Writer = esp32Port.writable.getWriter();
            esp32ReaderLoop();
        }

        camOpenIdx = classIdx;
        camSourceType = 'esp32';
        render();
        setStatus('🔌 ESP32-S3 camera connected — live view only (samples still captured via Webcam).');

        esp32RequestFrame();
        esp32CaptureTimer = setInterval(esp32RequestFrame, 1000);
    }catch(e){
        alert('ESP32-S3 connect error: '+e.message);
        console.error(e);
        await esp32Disconnect();
    }
}

function esp32RequestFrame(){
    if(camSourceType !== 'esp32' || esp32Busy || !esp32Writer) return;
    esp32Busy = true;
    if(esp32CaptureTimeout) clearTimeout(esp32CaptureTimeout);
    esp32CaptureTimeout = setTimeout(()=>{
        esp32Busy = false; esp32ImageActive = false;
        setStatus('ESP32-S3: capture timed out, retrying...');
    }, 5000);
    esp32Writer.write(new TextEncoder().encode('CAPTURE\n')).catch(e=>{
        esp32Busy = false;
        setStatus('ESP32-S3 write error: '+e.message);
    });
}

async function esp32ReaderLoop(){
    const decoder = new TextDecoder();
    try{
        while(esp32Port){
            esp32Reader = esp32Port.readable.getReader();
            try{
                while(true){
                    const {value,done} = await esp32Reader.read();
                    if(done) break;
                    if(value) esp32HandleText(decoder.decode(value,{stream:true}));
                }
            }finally{ esp32Reader.releaseLock(); }
        }
    }catch(e){
        if(esp32Port) console.error('ESP32-S3 serial read error:',e);
    }
}

function esp32HandleText(text){
    esp32LineBuf += text.replace(/\r/g,'');
    let idx;
    while((idx = esp32LineBuf.indexOf('\n')) >= 0){
        const line = esp32LineBuf.slice(0,idx);
        esp32LineBuf = esp32LineBuf.slice(idx+1);
        esp32ProcessLine(line);
    }
}

function esp32ProcessLine(line){
    if(line.startsWith('SERIAL IMAGE_BEGIN')){
        esp32ImageActive = true; esp32ImageB64 = '';
        return;
    }
    if(line.startsWith('SERIAL IMAGE_CHUNK ')){
        if(esp32ImageActive) esp32ImageB64 += line.slice('SERIAL IMAGE_CHUNK '.length).trim();
        return;
    }
    if(line.startsWith('SERIAL IMAGE_END')){
        if(esp32ImageActive && esp32ImageB64) esp32RenderFrame(esp32ImageB64);
        esp32ImageActive = false; esp32ImageB64 = ''; esp32Busy = false;
        if(esp32CaptureTimeout){clearTimeout(esp32CaptureTimeout); esp32CaptureTimeout=null;}
        return;
    }
    if(line.startsWith('SERIAL ERR')){
        esp32ImageActive = false; esp32ImageB64 = ''; esp32Busy = false;
        if(esp32CaptureTimeout){clearTimeout(esp32CaptureTimeout); esp32CaptureTimeout=null;}
        setStatus('ESP32-S3: '+line.replace(/^SERIAL ERR\s*/,''));
    }
}

function esp32RenderFrame(b64){
    if(camSourceType !== 'esp32' || camOpenIdx === null) return;
    const canvas = document.getElementById(`cam-canvas-${camOpenIdx}`);
    if(!canvas) return;
    const img = new Image();
    img.onload = ()=>{
        if(camSourceType !== 'esp32' || camOpenIdx === null) return;
        const liveCanvas = document.getElementById(`cam-canvas-${camOpenIdx}`);
        if(!liveCanvas) return;
        const ctx = liveCanvas.getContext('2d');
        ctx.clearRect(0,0,liveCanvas.width,liveCanvas.height);
        ctx.drawImage(img,0,0,liveCanvas.width,liveCanvas.height);
    };
    img.onerror = ()=>setStatus('ESP32-S3: failed to decode frame.');
    img.src = 'data:image/jpeg;base64,'+b64;
}

async function esp32Disconnect(){
    if(esp32CaptureTimer){clearInterval(esp32CaptureTimer); esp32CaptureTimer=null;}
    if(esp32CaptureTimeout){clearTimeout(esp32CaptureTimeout); esp32CaptureTimeout=null;}
    esp32Busy=false; esp32ImageActive=false; esp32ImageB64=''; esp32LineBuf='';
    const p = esp32Port; esp32Port = null;
    try{esp32Reader && await esp32Reader.cancel();}catch{}
    try{esp32Reader && esp32Reader.releaseLock();}catch{}
    try{esp32Writer && await esp32Writer.close();}catch{}
    try{esp32Writer && esp32Writer.releaseLock();}catch{}
    try{p && await p.close();}catch{}
    esp32Reader = esp32Writer = null;
}

    if(classes.length<=2){alert('Need at least 2 classes');return;}
    const cls = classes[idx];
    for(let i=dataset.length-1;i>=0;i--){if(dataset[i].label===idx)dataset.splice(i,1);}

    const newThumbnailStore = {};
